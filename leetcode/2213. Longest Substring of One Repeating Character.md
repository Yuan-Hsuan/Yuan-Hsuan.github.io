class Solution {
public:
    vector<int> longestRepeating(string s, string queryCharacters, vector<int>& queryIndices) {
        
        int n = s.length();
        set<pair<int, int>> segs;
        multiset<int> lens;

        for (int i = 0; i < n;) {
            int j = i;
            while (j < n && s[i] == s[j]) {
                j++;
            }

            segs.insert({i, j-1});
            lens.insert(j-i);
            i = j;
        }

        int k = queryIndices.size();
        vector<int> res(k);

        for (int q = 0; q < k; q++) {
            int pos = queryIndices[q];
            char ch = queryCharacters[q];

            if (s[pos] != ch) {
                
                auto it = segs.upper_bound({pos, INT_MAX});
                --it;
                int l = it->first, r = it->second;
                
                int newL = pos, newR = pos;
                if (pos > 0 && s[pos] != s[pos-1] && ch == s[pos-1]) {
                    auto prev = std::prev(it);
                    newL = prev->first;
                    lens.erase(lens.find(prev->second - prev->first + 1));

                    segs.erase(prev);
                }

                if (pos < n-1 && s[pos] != s[pos+1] && ch == s[pos+1]) {
                    auto next = std::next(it);
                    newR = next->second;
                    lens.erase(lens.find(next->second - next->first + 1));

                    segs.erase(next);
                }

                segs.erase(it);
                lens.erase(lens.find(r-l+1));

                if (pos > l) {
                    segs.insert({l, pos-1});
                    lens.insert(pos - l);
                }

                if (pos < r) {
                    segs.insert({pos+1, r});
                    lens.insert(r - pos);
                }

                segs.insert({newL, newR});
                lens.insert(newR - newL + 1);

                s[pos] = ch;
            }
            res[q] = *lens.rbegin();
        }

        return res;
    }
};
